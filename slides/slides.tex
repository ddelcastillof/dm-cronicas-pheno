\begin{frame}
  \frametitle{Introduction (i)}
  \begin{itemize}
    \item Cardiovascular disease is a global health threat \cite{Global-Burden-of-Cardiovascular-Diseases-and-Risks-2023-Collaborators2025-pr}, 
    projected to become one of the largest contributors to disease burden by 2100, particularly in LMICs \cite{Murray2026-np}.
    \item Type 2 diabetes (T2D) is one of its major contributors, while type 1 diabetes (T1D) classically was not \cite{Pinzon-Cortes2026-mv}. 
    \item However, diabetes is heterogeneous. Insulin resistance and autoimmunity overlap on a spectrum, and many individuals show features of both \cite{Redondo2020-fn, Franks2025-om}.
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Introduction (ii)}
  \begin{itemize}
    \item A data-driven framework from clinical parameters has been previously proposed in individuals from European ancestry, with different risk of micro and macrovascular complications \cite{Ahlqvist2018-su, Bennet2021-en}.
    \item A more informative categorization of diabetic patients will provide insigths on patient CVD stratification in resource-constrained settings.
  \end{itemize}
\end{frame}
%=====================================
% Methods
%=====================================
\begin{frame}
  \frametitle{Specific Aims}
  \begin{itemize}
    \item To classify individuals with type 2 diabetes in the CRONICAS cohort into Ahlqvist data-driven subtypes.
    \item To assess the transferability of a T2D polygenic risk score validated in a admixed mesoamerican sample to the Puno site of the CRONICAS cohort.
    \item To examine whether polygenic liability to T2D differs across subtypes in the Puno site of the CRONICAS cohort.
  \end{itemize}
\end{frame}


\begin{frame}
  \frametitle{Methods (i)}
  \begin{columns}[c, onlytextwidth]
    \begin{column}{0.5\textwidth}
      % half-width text overflows into the footer logo at 11pt; \small would also loosen beamer's list spacing
      \fontsize{10}{12}\selectfont
      \begin{itemize}
        \item \textbf{Aim 1:} CRONICAS cohort
          \begin{itemize}
            \item Prospective cohort study with 3135 participants at baseline \cite{Miranda2012-cg}.
            \item T2D prevalence of 7.1\% at baseline (223/3135) \cite{Bernabe-Ortiz2016-an, Carrillo-Larco2018-js}.
            \item Puno with the highest incidence at follow-up \cite{Bernabe-Ortiz2016-an}.
          \end{itemize}
        \item Supervised machine learning classification using coordinates from Ahlqvist et al \cite{Ahlqvist2018-su}.
        \item Stability will be assessed computing Jaccard similarity indexes with boostrap resampling (1000 reps).
        \item Sensitivity analysis: Gaussian mixture model.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \centering
      % Peru map with CRONICAS sites (Tumbes, Lima, Puno) shaded; made by assets/peru-map.py
      \includegraphics[height=5.5cm]{peru-map}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}
  \frametitle{Methods (ii)}
  \begin{itemize}
    \item \textbf{Aim 2:} Polygenic risk score transferability
      \begin{itemize}
        \item Torres et al.\cite{Torres2026-qs} conducted genomic analyses in 19,431 T2D cases from the Mexico City Prospective Study
        \item 86 GWAS associations were identified in this admixed mesoamerican population.
      \end{itemize}
    \item T2D polygenic scores will be computed in genotyped participants of the Puno site (n = 669) \cite{Castaneda2026-qq}.
    \item Transferability will be assessed by estimating the association of each score, with prevalent and incident diabetes and with continuous glycaemic traits.
    \item Performance will be reported as the score's incremental contribution.
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Methods (iii)}
  \begin{itemize}
    \item \textbf{Aim 3:} Compare polygenic liability across subtypes
    \item Test whether overall and process-specific T2D polygenic scores differ across Aim 1 subtypes
    \item Soft membership probabilities will be used to relate genetic signals to subtype assignment (exploratory, given sample size)
  \end{itemize}
\end{frame}
